-- V5 BRUTAL DELETE CANTER - TEST LANGSUNG
for _,v in pairs(game:GetService("Workspace"):GetDescendants()) do
    if v:IsA("BasePart") or v:IsA("MeshPart") then
        local n = string.lower(v.Name)
        -- Target pengaman & bumper
        if string.find(n, "pengaman") or string.find(n, "bumper") or string.find(n, "bullbar") or string.find(n, "bodykit") or n == "pengaman.001" or n == "pengaman.002" then
            pcall(function()
                v:Destroy()
            end)
        end
        -- Jika parent nya Pengaman
        if v.Parent and string.find(string.lower(v.Parent.Name), "pengaman") then
            pcall(function()
                v:Destroy()
            end)
        end
    end
end
print("DONE - PENGAMAN DESTROYED")

for _, v in pairs(W:GetDescendants()) do
    if v.Name == "Pengaman.001" or v.Name == "Pengaman" or v.Name == "Bumper" then
        if v:IsA("Model") then
            for _, p in pairs(v:GetDescendants()) do
                if p:IsA("BasePart") then
                    p.Transparency = 1
                    p.CanCollide = false
                end
            end
        else
            v.Transparency = 1
            v.CanCollide = false
        end
    end
end
