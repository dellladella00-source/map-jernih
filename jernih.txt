local W = game:GetService("Workspace")
local L = game:GetService("Lighting")

-- Map Jernih tetap
L.FogEnd = 100000
L.Brightness = 2

local targetNames = {
    "Pengaman", "pengaman", "Bumper", "bumper", "Bodykit", "Bullbar",
    "Front", "AddOn", "Lip", "Spoiler", "CanterFE74HD"
}

local function isTarget(obj)
    local n = obj.Name
    for _, t in pairs(targetNames) do
        if string.find(n, t) then
            return true
        end
    end
    -- Cek juga kalo dia anak dari Pengaman
    if obj.Parent and string.find(obj.Parent.Name, "Pengaman") then
        return true
    end
    return false
end

local function Hapus(obj)
    if obj:IsA("BasePart") or obj:IsA("MeshPart") then
        if isTarget(obj) then
            -- Jangan hapus ban
            if string.find(string.lower(obj.Name), "wheel") or string.find(string.lower(obj.Name), "ban") then return end
            
            obj.Transparency = 1
            obj.CanCollide = false
            if obj:FindFirstChildWhichIsA("Decal") then
                obj:FindFirstChildWhichIsA("Decal").Transparency = 1
            end
            -- Hapus semua texture
            for _,c in pairs(obj:GetChildren()) do
                if c:IsA("Texture") or c:IsA("Decal") then
                    c:Destroy()
                end
            end
        end
    end
end

-- Scan semua
for _, v in pairs(W:GetDescendants()) do
    Hapus(v)
end

W.DescendantAdded:Connect(function(n)
    task.wait(0.2)
    Hapus(n)
end)

-- Cara brutal kedua: cari model Canter nya langsung
for _, v in pairs(W:GetDescendants()) do
    if v.Name == "Pengaman.001" or v.Name == "Pengaman" or v.Name == "Bumper" then
        if v:IsA("Model") then
            for _, p in pairs(v:GetDescendants()) do
                if p:IsA("BasePart") then
                    p.Transparency = 1
                    p.CanCollide = false
                end
            end
        else
            v.Transparency = 1
            v.CanCollide = false
        end
    end
end

print("✅ V4 Canter Loaded - Pengaman.001 Hilang!")
