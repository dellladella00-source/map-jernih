-- Ban tetap ada, cuma bemper & pengaman samping yang hilang
local L = game:GetService("Lighting")
local W = game:GetService("Workspace")

-- 1. MAP JERNIH
L.FogEnd = 100000
L.FogStart = 0
L.Brightness = 2
L.GlobalShadows = false
L.ExposureCompensation = 0.1
for _,v in pairs(L:GetChildren()) do
    if v:IsA("BlurEffect") or v:IsA("SunRays") or v:IsA("ColorCorrectionEffect") or v:IsA("BloomEffect") or v:IsA("DepthOfFieldEffect") then
        v.Enabled = false
    end
end

-- 2. HAPUS BEMPER & PENGAMAN (ANTI BAN)
local function IsBumper(name)
    name = string.lower(name)
    -- Kalo ada kata ban/wheel/roda, JANGAN dihapus (return false)
    if string.find(name, "ban") or string.find(name, "wheel") or string.find(name, "tire") or string.find(name, "velg") or string.find(name, "roda") then
        return false
    end
    -- Yang dihapus cuma ini
    if string.find(name, "bumper") or string.find(name, "bemper") or string.find(name, "bullbar") or string.find(name, "bull") or string.find(name, "guard") or string.find(name, "pengaman") or string.find(name, "front") or string.find(name, "side") and string.find(name, "skirt") or string.find(name, "bodykit") or string.find(name, "mudguard") or string.find(name, "spakbor") then
        return true
    end
    -- Tambahan buat truk Canter biasanya namanya FrontLip, Lower, AddOn
    if name == "bumper" or name == "frontbumper" or name == "f_bumper" or name == "guard" then
        return true
    end
    return false
end

local function Hapus(obj)
    if obj:IsA("BasePart") and IsBumper(obj.Name) then
        obj.Transparency = 1
        obj.CanCollide = false
        -- Hapus juga decal nya biar bener2 hilang
        for _,d in pairs(obj:GetChildren()) do
            if d:IsA("Decal") or d:IsA("Texture") then
                d.Transparency = 1
            end
        end
    end
end

for _,v in pairs(W:GetDescendants()) do
    Hapus(v)
end

W.DescendantAdded:Connect(function(n)
    task.wait(0.3)
    Hapus(n)
end)

print("✅ V3 Loaded - Bumper & Pengaman Hilang, Ban Aman!")
