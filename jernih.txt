local Lighting = game:GetService("Lighting")
local Workspace = game:GetService("Workspace")

-- 1. MAP JADI JERNIH
Lighting.FogEnd = 100000
Lighting.FogStart = 0
Lighting.Brightness = 2
Lighting.GlobalShadows = false
for i,v in pairs(Lighting:GetChildren()) do
    if v:IsA("BlurEffect") or v:IsA("SunRays") or v:IsA("ColorCorrectionEffect") or v:IsA("BloomEffect") or v:IsA("DepthOfFieldEffect") then
        v.Enabled = false
    end
end

-- 2. HAPUS BAN & AKSESORIS
local function HapusPart(obj)
    local nama = string.lower(obj.Name)
    if string.find(nama, "ban") or string.find(nama, "wheel") or string.find(nama, "tire") or string.find(nama, "velg") or string.find(nama, "accessory") or string.find(nama, "spion") then
        if obj:IsA("BasePart") then
            obj.Transparency = 1
        end
    end
end

for _, v in pairs(Workspace:GetDescendants()) do
    HapusPart(v)
end

Workspace.DescendantAdded:Connect(function(newObj)
    task.wait(0.5)
    HapusPart(newObj)
end)

print("✅ Map Jernih Loaded - dellladella00-source")
