local Players = game:GetService("Players")
local Workspace = game:GetService("Workspace")
local Lighting = game:GetService("Lighting")
local TweenService = game:GetService("TweenService")
local VIM = game:GetService("VirtualInputManager")
local LP = Players.LocalPlayer
local RS = game:GetService("ReplicatedStorage")

print("🔥 LOADING V2 SULTAN ALL MAP...")

-- CONFIG SULTAN
getgenv().Config = {
    MapJernih = true,
    HapusBemper = true,
    AutoFarm = true,
    JobTermahalOnly = true, -- Cuma ambil job mahal
    Kecepatan = 70 -- Kecepatan realistis 70km/jam biar gak ke-detect
}

-- 1. MAP JERNIH V2 (ALL MAP 100% WORK)
if getgenv().Config.MapJernih then
    Lighting.FogEnd = 100000
    Lighting.FogStart = 0
    Lighting.Brightness = 2.5
    Lighting.ClockTime = 14
    Lighting.GlobalShadows = false
    Lighting.OutdoorAmbient = Color3.fromRGB(255,255,255)
    for _,v in pairs(Lighting:GetChildren()) do
        if v:IsA("PostEffect") then v.Enabled = false end
    end
    -- Hapus kabut di map apapun
    for _,v in pairs(Workspace:GetDescendants()) do
        if v:IsA("BasePart") and v.Name:lower():find("fog") then
            v:Destroy()
        end
    end
end

-- 2. HAPUS BEMPER + PENGAMAN (Biar Truk Enteng & Realistis)
if getgenv().Config.HapusBemper then
    local function FilterBemper(obj)
        if obj:IsA("BasePart") or obj:IsA("MeshPart") then
            local name = obj.Name:lower()
            if name:find("pengaman") or name:find("bemper") or name:find("bumper") or name:find("besi") then
                pcall(function()
                    obj.Transparency = 1
                    obj.CanCollide = false
                    obj.Massless = true
                end)
            end
        end
    end
    for _,v in pairs(Workspace:GetDescendants()) do FilterBemper(v) end
    Workspace.DescendantAdded:Connect(FilterBemper)
end

-- 3. ENGINE AUTO FARM V2 - REALISTIS & TERMHAL
local function GetMyTruck()
    for _,v in pairs(Workspace.Vehicles or Workspace:GetDescendants()) do
        if v:IsA("Model") and v:FindFirstChild("Owner") and v.Owner.Value == LP.Name then
            return v
        end
    end
    for _,v in pairs(Workspace:GetDescendants()) do
        if v:IsA("VehicleSeat") and v.Occupant and v.Occupant.Parent == LP.Character then
            return v.Parent
        end
    end
    return nil
end

local function GetBestJob()
    -- Cari job dengan bayaran tertinggi di map apapun
    local bestJob = nil
    local bestPrice = 0
    
    -- Scan semua job di workspace
    for _,v in pairs(Workspace:GetDescendants()) do
        if v:IsA("Folder") and (v.Name == "Jobs" or v.Name == "Missions") then
            for _,job in pairs(v:GetChildren()) do
                local price = job:GetAttribute("Price") or job:GetAttribute("Reward") or 0
                if price > bestPrice then
                    bestPrice = price
                    bestJob = job
                end
            end
        end
        -- Fallback: cari part job
        if v:IsA("Part") and v:FindFirstChild("TouchInterest") and v.Name:lower():find("job") then
            local price = v:GetAttribute("Price") or 5000
            if price > bestPrice then
                bestPrice = price
                bestJob = v
            end
        end
    end
    return bestJob, bestPrice
end

local function RealisticDrive(targetCFrame)
    local truck = GetMyTruck()
    if not truck then print("❌ Truk gak ketemu, naik truk dulu bos!") return end
    
    local seat = truck:FindFirstChildWhichIsA("VehicleSeat", true)
    if not seat then return end
    
    -- Drive realistis: gak langsung teleport, tapi tween pelan
    local distance = (seat.Position - targetCFrame.Position).Magnitude
    local duration = distance / getgenv().Config.Kecepatan
    
    local tween = TweenService:Create(seat, TweenInfo.new(duration, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut), {CFrame = targetCFrame})
    tween:Play()
    tween.Completed:Wait()
end

-- LOOP UTAMA FARM SULTAN
spawn(function()
    while getgenv().Config.AutoFarm do
        local job, price = GetBestJob()
        
        if job then
            print("💰 NEMU JOB SULTAN! Harga: Rp "..price.." | Map: "..game.PlaceId)
            print("📍 Lokasi: "..job:GetFullName())
            
            -- Jalan realistis ke job
            local targetPos = job:IsA("Part") and job.CFrame or job:GetPivot()
            RealisticDrive(targetPos * CFrame.new(0,5,0))
            
            task.wait(1)
            -- Ambil job (pencet E realistis)
            VIM:SendKeyEvent(true, "E", false, game)
            task.wait(0.3)
            VIM:SendKeyEvent(false, "E", false, game)
            
            task.wait(3)
            
            -- Cari tempat antar
            for _,v in pairs(Workspace:GetDescendants()) do
                if v.Name:lower():find("antar") or v.Name:lower():find("delivery") or v.Name:lower():find("drop") then
                    if v:IsA("Part") then
                        print("🚚 Antar ke tujuan...")
                        RealisticDrive(v.CFrame)
                        VIM:SendKeyEvent(true, "E", false, game)
                        task.wait(0.3)
                        VIM:SendKeyEvent(false, "E", false, game)
                        break
                    end
                end
            end
        else
            print("⏳ Map ID "..game.PlaceId.." | Cari job termahal... muter dulu bos")
            task.wait(5)
        end
        task.wait(2)
    end
end)
