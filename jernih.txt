--- V5 SUPER BRUTAL - HAPUS MODEL PENGAMAN LANGSUNG
local ws = workspace
for _,m in pairs(ws:GetDescendants()) do
    if m.Name == "Pengaman.001" then
        m:Destroy()
    end
    if m.Name == "CanterFE74HD" then
        -- Jangan hapus semua, cuma anaknya yang pengaman
        for _,c in pairs(m:GetDescendants()) do
            if string.lower(c.Name):find("pengaman") then
                c:Destroy()
            end
        end
    end
endend

print("✅ V4 Canter Loaded - Pengaman.001 Hilang!")
