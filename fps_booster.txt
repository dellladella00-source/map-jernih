-- ALL MAP RINGAN - HP KENTANG EDITION
-- Cuma ngerubah tampilan di HP lu, gak ngerubah map orang

local Lighting = game:GetService("Lighting")
local Terrain = workspace:FindFirstChildOfClass("Terrain")

-- 1. MATIKAN SEMUA EFEK BERAT
Lighting.GlobalShadows = false
Lighting.FogEnd = 9e9
Lighting.ShadowDensity = 0
Lighting.EnvironmentDiffuseScale = 0
Lighting.EnvironmentSpecularScale = 0
settings().Rendering.QualityLevel = 1
UserSettings():GetService("UserGameSettings").SavedQualityLevel = 1

-- 2. HAPUS EFEK DI LAYAR LU DOANG
for _, v in pairs(game:GetDescendants()) do
    if v:IsA("ParticleEmitter") or v:IsA("Trail") or v:IsA("Beam") or v:IsA("Smoke") or v:IsA("Fire") or v:IsA("Sparkles") then
        v.Enabled = false
    end
    if v:IsA("BasePart") then
        v.CastShadow = false
        v.Material = Enum.Material.SmoothPlastic
    end
    if v:IsA("Decal") or v:IsA("Texture") then
        v.Transparency = 1
    end
end

-- 3. RENDER JARAK DEKET DOANG
if Terrain then
    Terrain.WaterWaveSize = 0
    Terrain.WaterWaveSpeed = 0
    Terrain.WaterReflectance = 0
    Terrain.WaterTransparency = 0
end

workspace.DescendantAdded:Connect(function(v)
    if v:IsA("ParticleEmitter") or v:IsA("Trail") or v:IsA("Beam") then
        v.Enabled = false
    end
end)

print("✅ ALL MAP JADI RINGAN DI HP LU BOS - ANTI NGELAG")
